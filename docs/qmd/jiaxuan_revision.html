  sprintf("%.2f (%.2f-%.2f)", exp(delta * b),
          exp(delta * b - 1.96 * abs(delta) * se),
          exp(delta * b + 1.96 * abs(delta) * se))

.prep <- function(o, xvar) {
  adj <- c(adj_base, if (!is.na(o$extra)) o$extra)
  d <- dat %>%
    dplyr::select(dplyr::all_of(c(xvar, o$time, o$status, adj))) %>%
    tidyr::drop_na()
  list(d = d, adj = adj)
}

.fit_terms <- function(o, d, adj, Xexp) {
  k <- ncol(Xexp)
  if (o$model == "fg") {
    X_adj  <- model.matrix(~ ., data = d[, adj, drop = FALSE])[, -1, drop = FALSE]
    status <- suppressWarnings(as.integer(as.character(d[[o$status]])))
    fit <- cmprsk::crr(ftime = as.numeric(d[[o$time]]), fstatus = status,
                       cov1 = cbind(Xexp, X_adj), failcode = 1)
    list(beta = fit$coef[seq_len(k)],
         V = fit$var[seq_len(k), seq_len(k), drop = FALSE],
         n = nrow(d), events = sum(status == 1))
  } else {
    nm  <- colnames(Xexp)
    fit <- survival::coxph(
      as.formula(paste0("survival::Surv(", o$time, ", ", o$status, ") ~ ",
                        paste(c(nm, adj), collapse = " + "))),
      data = cbind(d, Xexp))
    list(beta = coef(fit)[nm], V = vcov(fit)[nm, nm, drop = FALSE],
         n = nrow(d), events = sum(d[[o$status]] == 1))
  }
}

fit_linear <- function(o, xvar) {
  p <- .prep(o, xvar)
  X <- matrix(p$d[[xvar]], ncol = 1, dimnames = list(NULL, "x_lin"))
  f <- .fit_terms(o, p$d, p$adj, X)
  b <- unname(f$beta[1]); se <- sqrt(f$V[1, 1])
  c(f, list(b = b, se = se, x = p$d[[xvar]], p = 2 * pnorm(-abs(b / se))))
}

rcs_mat <- function(x, knots) {
  B <- Hmisc::rcspline.eval(x, knots = knots, inclx = TRUE)
  B <- matrix(B, nrow = length(x))
  colnames(B) <- paste0("rcs", seq_len(ncol(B)))
  B
}

fit_spline <- function(o, xvar) {
  p  <- .prep(o, xvar)
  x  <- p$d[[xvar]]
  kn <- as.numeric(quantile(x, c(0.05, 0.35, 0.65, 0.95)))
  f  <- .fit_terms(o, p$d, p$adj, rcs_mat(x, kn))
  bn <- f$beta[-1]; Vn <- f$V[-1, -1, drop = FALSE]
  chi_nl  <- as.numeric(t(bn) %*% solve(Vn, bn))
  chi_all <- as.numeric(t(f$beta) %*% solve(f$V, f$beta))
  c(f, list(x = x, knots = kn,
            p_nl  = pchisq(chi_nl,  df = length(bn),     lower.tail = FALSE),
            p_all = pchisq(chi_all, df = length(f$beta), lower.tail = FALSE)))
}

plot_linear_curve <- function(key, fits = fits_lin_log, xlab = xlab_log,
                              scale_txt = " (log scale)", iqr_lines = FALSE,
                              n_bonf = 6) {
  o <- outcomes[outcomes$key == key, ]
  f <- fits[[key]]
  x <- f$x
  ref  <- median(x)
  grid <- seq(quantile(x, 0.025), quantile(x, 0.975), length.out = 200)
  lhr  <- f$b * (grid - ref)
  se   <- abs(grid - ref) * f$se
  q1 <- unname(quantile(x, 0.25)); q3 <- unname(quantile(x, 0.75)); iqr <- q3 - q1
  hr_lab <- if (o$model == "cox") "HR" else "subHR"
  subtitle_str <- sprintf(
    "%s per 10 %% decline%s: %s, p_Bonf = %s (n = %d)\nper IQR decline (Δ = %.1f): %s  |  IQR: [%.1f, %.1f]",
    hr_lab, scale_txt, fmt_hr(f$b, f$se, -10), fmt_p(min(f$p * n_bonf, 1)), n_bonf,
    iqr, fmt_hr(f$b, f$se, -iqr), q1, q3)
  p <- ggplot(tibble(x = grid, hr = exp(lhr),
                     lcl = exp(lhr - 1.96 * se), ucl = exp(lhr + 1.96 * se)),
              aes(x = x, y = hr)) +
    geom_ribbon(aes(ymin = lcl, ymax = ucl), alpha = 0.2, fill = "darkorange") +
    geom_line(color = "darkorange", linewidth = 1) +
    geom_hline(yintercept = 1, linetype = "dashed", color = "grey50") +
    geom_vline(xintercept = ref, linetype = "dotted", color = "grey40")
  if (iqr_lines) {
    p <- p + geom_vline(xintercept = c(q1, q3), linetype = "dashed",
                        color = "steelblue", alpha = 0.7)
  }
  p +
    scale_x_continuous(breaks = seq(-200, 200, 10)) +
    coord_cartesian(ylim = c(0, 4)) +
    labs(x = xlab, y = paste0(hr_lab, " (95% CI)"),
         title = o$label, subtitle = subtitle_str) +
    theme_minimal(base_size = 11) +
    theme(plot.subtitle = element_text(size = 9, color = "grey25"))
}

spline_vs_linear <- function(key) {
  fs <- fits_rcs_log[[key]]
  fl <- fits_lin_log[[key]]
  x  <- fs$x
  ref  <- median(x)
  grid <- seq(quantile(x, 0.025), quantile(x, 0.975), length.out = 200)
  D    <- sweep(rcs_mat(grid, fs$knots), 2, rcs_mat(ref, fs$knots)[1, ])
  lhr  <- as.numeric(D %*% fs$beta)
  se   <- sqrt(rowSums((D %*% fs$V) * D))
  tibble(x = grid, ref = ref,
         spline = exp(lhr), lcl = exp(lhr - 1.96 * se), ucl = exp(lhr + 1.96 * se),
         linear = exp(fl$b * (grid - ref)))
}

3 eGFR difference on the log scale (reviewer item 1)

3.1 Why the log scale

The submitted metric, 100 × (CKD-EPI − CG) / CG, divides by CG, which makes it asymmetric: if CKD-EPI Cr-Cys is 20 % below CG, then CG is 25 % above CKD-EPI Cr-Cys, and the scale is bounded at −100 % but unbounded above. Expressing the difference as 100 × ln(CKD-EPI Cr-Cys / CG) (the “symmetric percentage”, Cole, Stat Med 2000;19:3109–25; Törnqvist et al., Am Stat 1985;39:43–6) has three advantages:

  1. Symmetric with respect to the reference. Swapping numerator and denominator only flips the sign (+22.3 ↔︎ −22.3), so the result does not depend on treating CG as the reference.
  2. Additive. Log differences sum across steps, and the metric is the log of the ratio of the two GFR estimates, which is the natural scale for comparing them.
  3. Same interpretation for small differences. Within about ±15 % the two metrics are almost identical (Table 3.3). A 10-unit decline on the log scale corresponds to a CKD-EPI/CG ratio that is 9.5 % lower (ratio × e−0.1 = 0.905).

Note for the response letter: in this cohort the log transformation does not make the distribution more symmetric (skewness -0.44 on the log scale vs 0.20 for the submitted metric); it lengthens the lower tail. The justification therefore rests on the properties of the measure (points 1–3), not on its distribution.

4 Linearity of the sHR curves (reviewer item 2)

Reviewer: “In the figures where sHR were plotted, did you force linearity of eGFR difference or are these based on splines to capture potential non-linearity of these associations?”

4.1 What the submitted figures show

Linearity was assumed, not estimated. Each submitted curve came from a Fine-Gray model (Cox proportional hazards for 90-day death) in which eGFR difference entered as a single linear term together with the full adjuster set. The plotted line is sHR(x) = exp{β (x − median)} with a 95 % delta-method confidence band, so it is a straight line on the log-sHR scale by construction: the figures show the size of a linear association across the observed range (2.5th–97.5th percentile), not its shape.

4.2 Testing for non-linearity

Every model was re-fitted with eGFR difference modelled as a restricted cubic spline with 4 knots at the 5th, 35th, 65th and 95th percentiles (Harrell’s default placement), keeping the covariates, the competing-risk handling and the analytic cohort unchanged. Departure from linearity was tested with a Wald test of the two non-linear spline coefficients (2 df), using the Fine-Gray sandwich variance from cmprsk::crr (model-based variance for Cox). The same test was run on the submitted % metric. p-values are unadjusted. The “inside spline 95% CI” column gives the share of the plotted range (2.5th–97.5th percentile) over which the linear curve lies within the spline’s confidence band.

5 BSA-indexed eGFR difference (%) (item 3)

5.1 Definition

Definition requested: (eGFRCR-CYS in mL/min/1.73 m², not adjusted to the patient’s BSA − CG CrCl indexed to 1.73 m²) / (CG CrCl indexed to 1.73 m²), i.e.

egfr_diff_pct_bsa = 100 × (ckd_epi_gfr_cre_cys − CG × 1.73/BSA) / (CG × 1.73/BSA)

ImportantThis is the same number as the submitted % difference

Dividing the numerator and denominator by 1.73/BSA cancels BSA: (eGFRCR-CYS per 1.73 m² − CG × 1.73/BSA) / (CG × 1.73/BSA) = (eGFRCR-CYS per 1.73 m² × BSA/1.73 − CG) / CG = (eGFRCR-CYS unindexed − CG) / CG.

The unindexed CKD-EPI value used throughout the analysis is the indexed value × BSA/1.73 with the same DuBois BSA, so the two definitions agree for every patient (largest absolute difference across the cohort: 3.6e-14 percentage points, i.e. floating-point rounding). Indexing to body size changes the absolute difference in mL/min, but not the percent difference, so all models and figures using the submitted % metric are unchanged under this definition.
